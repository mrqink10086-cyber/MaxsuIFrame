cmake_minimum_required(VERSION 3.21)
project(MaxsuIFrameTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(frames_tests frames_tests.cpp ../src/InvincibleFrames.cpp)
target_include_directories(frames_tests PRIVATE ../src)
target_link_libraries(frames_tests PRIVATE nlohmann_json::nlohmann_json)
if(MSVC)
	target_compile_options(frames_tests PRIVATE /W4 /permissive- /utf-8)
	option(ENABLE_ASAN "Run the standalone tests with AddressSanitizer" OFF)
	if(ENABLE_ASAN)
		target_compile_options(frames_tests PRIVATE /fsanitize=address /Zi)
		target_link_options(frames_tests PRIVATE /INCREMENTAL:NO /DEBUG)
	endif()
endif()
enable_testing()
add_test(NAME frames_tests COMMAND frames_tests)
